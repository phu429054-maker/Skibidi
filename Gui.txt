local function getGradColor(g, isEnd)
	local ok, seq = pcall(function() return g.Color end)
	if not ok or not seq then return Color3.fromRGB(255,255,255) end
	local kp = seq.Keypoints
	return (isEnd and kp[#kp] or kp[1]).Value
end

local function setGradColor(g, isEnd, c)
	local ok, seq = pcall(function() return g.Color end)
	if not ok or not seq then return end
	local kp = seq.Keypoints
	local nk = {}
	for i, p in ipairs(kp) do
		local t = (isEnd and i == #kp) or ((not isEnd) and i == 1)
		table.insert(nk, ColorSequenceKeypoint.new(p.Time, t and c or p.Value))
	end
	g.Color = ColorSequence.new(nk)
end

local function connectRB(frame)
	if not frame or not frame:IsA("GuiObject") then return end
	if rbConns[frame] then return end
	local g = frame:FindFirstChildOfClass("UIGradient")
	if not g then g = Instance.new("UIGradient") g.Rotation = 45 g.Parent = frame end
	frame:SetAttribute("RainbowEnabled", true)
	local o = 0
	rbConns[frame] = RS.Heartbeat:Connect(function(dt)
		if not frame or not frame.Parent then
			if rbConns[frame] then rbConns[frame]:Disconnect() rbConns[frame] = nil end
			return
		end
		if not g or not g.Parent then
			g = frame:FindFirstChildOfClass("UIGradient")
			if not g then return end
		end
		o = (o + dt * 0.3) % 1
		local k = {}
		for i = 0, 7 do
			local t = i / 7
			table.insert(k, ColorSequenceKeypoint.new(t, Color3.fromHSV((t + o) % 1, 1, 1)))
		end
		g.Color = ColorSequence.new(k)
	end)
end

local function disconnectAllRB()
	for _, c in pairs(rbConns) do pcall(function() c:Disconnect() end) end
	rbConns = {}
end

local function attachPrevDrag(gui)
	local d, ds, sp = false, nil, nil
	local c1 = gui.InputBegan:Connect(function(i)
		if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
			d = true ds = i.Position sp = gui.Position
		end
	end)
	local c2 = UIS.InputChanged:Connect(function(i)
		if d and (i.UserInputType == Enum.UserInputType.MouseMovement or i.UserInputType == Enum.UserInputType.Touch) then
			if not gui.Parent then d = false return end
			local dl = i.Position - ds
			gui.Position = UDim2.new(sp.X.Scale, sp.X.Offset + dl.X, sp.Y.Scale, sp.Y.Offset + dl.Y)
		end
	end)
	local c3 = UIS.InputEnded:Connect(function(i)
		if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then d = false end
	end)
	table.insert(prevDragConns, c1)
	table.insert(prevDragConns, c2)
	table.insert(prevDragConns, c3)
end

local function clearPrevDrag()
	for _, c in ipairs(prevDragConns) do pcall(function() c:Disconnect() end) end
	prevDragConns = {}
end

local function enterPrev()
	local ok, err = pcall(function()
		local code = genLua()
		local lf = loadstring or load
		if not lf then error("Executor không hỗ trợ loadstring") end
		local fn, se = lf(code)
		if not fn then error(se or "Syntax error") end
		hiddenGuis = {}
		for _, c in ipairs(UserFolder:GetDescendants()) do
			if c:IsA("ScreenGui") then
				table.insert(hiddenGuis, {gui=c, en=c.Enabled})
				c.Enabled = false
			end
		end
		local s, r = pcall(fn)
		if not s then
			for _, i in ipairs(hiddenGuis) do if i.gui and i.gui.Parent then i.gui.Enabled = i.en end end
			hiddenGuis = {}
			error(tostring(r))
		end
		prevGui = r
		isPrev = true
		G.explorer.Visible = false
		G.props.Visible = false
		if selFrame then selFrame.Visible = false end
		prevBtn.Text = "Edit Mode"
		prevBtn.BackgroundColor3 = Color3.fromRGB(0,150,70)
	end)
	if not ok then showErr(tostring(err)) end
end

	if inst:IsA("ImageLabel") or inst:IsA("ImageButton") then for _, p in ipairs(PS.Image) do table.insert(l, p) end end
	if inst:IsA("UICorner") then for _, p in ipairs(PS.UICorner) do table.insert(l, p) end end
	if inst:IsA("UIStroke") then for _, p in ipairs(PS.UIStroke) do table.insert(l, p) end end
	if inst:IsA("UIGradient") then for _, p in ipairs(PS.UIGradient) do table.insert(l, p) end end
	if inst:IsA("UIPadding") then for _, p in ipairs(PS.UIPadding) do table.insert(l, p) end end
	if inst:IsA("UIAspectRatioConstraint") then for _, p in ipairs(PS.UIAspect) do table.insert(l, p) end end
	if inst:IsA("UIDragDetector") then for _, p in ipairs(PS.UIDrag) do table.insert(l, p) end end
	if inst:IsA("LocalScript") then for _, p in ipairs(PS.LocalScript) do table.insert(l, p) end end
	if inst:IsA("ModuleScript") then for _, p in ipairs(PS.ModuleScript) do table.insert(l, p) end end

	if inst:IsA("VideoFrame") then for _, p in ipairs(PS.Video) do table.insert(l, p) end end

local function serialize(inst)
	local d = {ClassName=inst.ClassName, Name=inst.Name, Properties={}, Children={}}
	for _, pr in ipairs(getSchema(inst)) do
		if pr.Name ~= "DragGui" then
			local ok, v = pcall(function() return inst[pr.Name] end)
			if ok then
				if pr.Type == "Color3" then d.Properties[pr.Name] = {Type="Color3", R=v.R, G=v.G, B=v.B}
				elseif pr.Type == "UDim2" then d.Properties[pr.Name] = {Type="UDim2", XS=v.X.Scale, XO=v.X.Offset, YS=v.Y.Scale, YO=v.Y.Offset}
				elseif pr.Type == "UDim" then d.Properties[pr.Name] = {Type="UDim", S=v.Scale, O=v.Offset}
				elseif pr.Type == "Vector2" then d.Properties[pr.Name] = {Type="Vector2", X=v.X, Y=v.Y}
				else d.Properties[pr.Name] = v end
			end
		end
	end
	if inst:IsA("UIGradient") then
		local sc = getGradColor(inst, false)
		local ec = getGradColor(inst, true)
		d.Properties.GradientColorStart = {Type="Color3", R=sc.R, G=sc.G, B=sc.B}
		d.Properties.GradientColorEnd = {Type="Color3", R=ec.R, G=ec.G, B=ec.B}
	end
	if inst:GetAttribute("RainbowEnabled") then d.RainbowEnabled = true end
	if inst:GetAttribute("DragGuiEnabled") then d.DragGuiEnabled = true end
	for _, c in ipairs(inst:GetChildren()) do
		if c.Name ~= "SelectionBoxOverlay" and c.Name ~= "DragScript" and c.Name ~= "RainbowScript" then
			table.insert(d.Children, serialize(c))
		end
	end
	return d
end

local function deserialize(d, parent)
	local inst = Instance.new(d.ClassName)
	inst.Name = d.Name
	for pn, v in pairs(d.Properties) do
		if pn ~= "GradientColorStart" and pn ~= "GradientColorEnd" then
			pcall(function()
				if type(v) == "table" and v.Type then
					if v.Type == "Color3" then inst[pn] = Color3.new(v.R, v.G, v.B)
					elseif v.Type == "UDim2" then inst[pn] = UDim2.new(v.XS, v.XO, v.YS, v.YO)
					elseif v.Type == "UDim" then inst[pn] = UDim.new(v.S, v.O)
					elseif v.Type == "Vector2" then inst[pn] = Vector2.new(v.X, v.Y) end
				else inst[pn] = v end
			end)
		end
	end
	if inst:IsA("UIGradient") then
		local sd = d.Properties.GradientColorStart
		local ed = d.Properties.GradientColorEnd
		if sd then setGradColor(inst, false, Color3.new(sd.R, sd.G, sd.B)) end
		if ed then setGradColor(inst, true, Color3.new(ed.R, ed.G, ed.B)) end
	end
	inst.Parent = parent
	if d.RainbowEnabled and inst:IsA("GuiObject") then connectRB(inst) end
	if d.DragGuiEnabled and inst:IsA("GuiObject") then
		inst:SetAttribute("DragGuiEnabled", true)
	end
	for _, cd in ipairs(d.Children or {}) do deserialize(cd, inst) end
	return inst
end

	L[#L+1] = "local _rbFuncs = {}"
	local cnt = 0
	local function proc(inst, parentVar)
		cnt = cnt + 1
		local v = "HTT03[\""..inst.ClassName.."_"..cnt.."\"]"
		L[#L+1] = v.." = Instance.new(\""..inst.ClassName.."\", "..parentVar..")"
		L[#L+1] = v..".Name = \""..esc(inst.Name).."\""
		for _, pr in ipairs(getSchema(inst)) do
			if pr.Name ~= "Name" and pr.Name ~= "DragGui" then
				local ok, val = pcall(function() return inst[pr.Name] end)
				if ok and val ~= nil then
					if pr.Type == "string" or pr.Type == "CodeText" then
						L[#L+1] = v.."."..pr.Name.." = \""..esc(val).."\""
					elseif pr.Type == "boolean" then

						L[#L+1] = string.format("%s.%s = UDim2.new(%f,%d,%f,%d)", v, pr.Name, val.X.Scale, val.X.Offset, val.Y.Scale, val.Y.Offset)

					elseif pr.Type == "Vector2" then
						L[#L+1] = string.format("%s.%s = Vector2.new(%f,%f)", v, pr.Name, val.X, val.Y)
					end
				end
			end
		end

			local ec = getGradColor(inst, true)
			L[#L+1] = string.format("%s.Color = ColorSequence.new(Color3.fromRGB(%d,%d,%d), Color3.fromRGB(%d,%d,%d))", v, math.floor(sc.R*255), math.floor(sc.G*255), math.floor(sc.B*255), math.floor(ec.R*255), math.floor(ec.G*255), math.floor(ec.B*255))
		end
		if inst:GetAttribute("RainbowEnabled") and inst:IsA("GuiObject") then
			L[#L+1] = "do"
			L[#L+1] = "    local _f = "..v
			L[#L+1] = "    local _g = _f:FindFirstChildOfClass(\"UIGradient\")"

			L[#L+1] = "        _g.Color = ColorSequence.new(k)"
			L[#L+1] = "    end)"
			L[#L+1] = "end"
		end
		if inst:GetAttribute("DragGuiEnabled") and inst:IsA("GuiObject") then

			L[#L+1] = "    local _f = "..v
			L[#L+1] = "    local _d, _ds, _sp = false, nil, nil"

			L[#L+1] = "        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then _d = true; _ds = input.Position; _sp = _f.Position end"
			L[#L+1] = "    end)"
			L[#L+1] = "    UserInputService.InputChanged:Connect(function(input)"
			L[#L+1] = "        if _d and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then"
			L[#L+1] = "            if not _f.Parent then _d = false; return end"
			L[#L+1] = "            local delta = input.Position - _ds"
			L[#L+1] = "            _f.Position = UDim2.new(_sp.X.Scale, _sp.X.Offset + delta.X, _sp.Y.Scale, _sp.Y.Offset + delta.Y)"
			L[#L+1] = "        end"
			L[#L+1] = "    end)"
			L[#L+1] = "    UserInputService.InputEnded:Connect(function(input)"

			L[#L+1] = "    end)"

local function parseEnum(str, et)
	if str == "" or not et then return nil end
	local ok, items = pcall(function() return et:GetEnumItems() end)
	if not ok or not items then return nil end
	for _, m in pairs(items) do
		if string.lower(m.Name) == string.lower(str) then return m end
	end
	return nil
end
local function pU2(s)
	local n = {}
	for num in string.gmatch(s, "[%-%d%.]+") do table.insert(n, tonumber(num)) end
	if #n >= 4 then return UDim2.new(n[1], n[2], n[3], n[4]) end
	return nil
end
local function pV2(s)
	local n = {}
	for num in string.gmatch(s, "[%-%d%.]+") do table.insert(n, tonumber(num)) end
	if #n >= 2 then return Vector2.new(n[1], n[2]) end
	return nil
end
local function pU(s)
	local n = {}
	for num in string.gmatch(s, "[%-%d%.]+") do table.insert(n, tonumber(num)) end
	if #n >= 2 then return UDim.new(n[1], n[2]) end
	return nil
end

local function updFast(obj)
	if not obj or not selObj then return end
	if propInputs["Position"] then
		propInputs["Position"].Text = string.format("{%.2f, %d}, {%.2f, %d}", obj.Position.X.Scale, obj.Position.X.Offset, obj.Position.Y.Scale, obj.Position.Y.Offset)
	end
	if propInputs["Size"] then
		propInputs["Size"].Text = string.format("{%.2f, %d}, {%.2f, %d}", obj.Size.X.Scale, obj.Size.X.Offset, obj.Size.Y.Scale, obj.Size.Y.Offset)
	end
end

local function makeDrag(gui)
	local d, ds, sp = false, nil, nil
	gui.InputBegan:Connect(function(i)
		if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
			if selObj == gui then
				d = true ds = i.Position sp = gui.Position
				G.explorer.Visible = false G.props.Visible = false
			end
		end
	end)
	UIS.InputChanged:Connect(function(i)
		if d and (i.UserInputType == Enum.UserInputType.MouseMovement or i.UserInputType == Enum.UserInputType.Touch) then
			local dl = i.Position - ds
			gui.Position = UDim2.new(sp.X.Scale, sp.X.Offset + dl.X, sp.Y.Scale, sp.Y.Offset + dl.Y)
			updFast(gui)
		end
	end)
	local function stop(i)
		if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
			d = false G.explorer.Visible = not isPrev G.props.Visible = not isPrev
		end
	end
	gui.InputEnded:Connect(stop) UIS.InputEnded:Connect(stop)
end

local function showMenu(parent, pos)
	clearMenus()
	local mf = Instance.new("Frame", G.gui)
	mf.Size = UDim2.new(0,160,0,200) mf.Position = pos
	mf.BackgroundColor3 = Color3.fromRGB(35,35,35) mf.BorderSizePixel = 0 mf.ZIndex = 10
	local ms = Instance.new("ScrollingFrame", mf)
	ms.Size = UDim2.new(1,0,1,0) ms.BackgroundTransparency = 1 ms.BorderSizePixel = 0
	ms.ScrollBarThickness = 4 ms.AutomaticCanvasSize = Enum.AutomaticSize.Y
	local ml = Instance.new("UIListLayout", ms) ml.SortOrder = Enum.SortOrder.LayoutOrder

	if clipObj then
		local pb = Instance.new("TextButton", ms)
		pb.Size = UDim2.new(1,0,0,22) pb.BackgroundColor3 = Color3.fromRGB(0,120,215)
		pb.TextColor3 = Color3.fromRGB(255,255,255) pb.Text = "  Paste" pb.TextXAlignment = Enum.TextXAlignment.Left
		pb.Font = Enum.Font.SourceSansBold pb.TextSize = 13
		pb.MouseButton1Click:Connect(function()
			clearMenus()
			if clipObj and clipObj.Parent then
				if isCut then clipObj.Parent = parent clipObj = nil isCut = false
				else local p = clipObj:Clone() p.Parent = parent end
				renderTree()
			end
		end)
	end

	for _, cn in ipairs(CREATABLE) do
		local b = Instance.new("TextButton", ms)
		b.Size = UDim2.new(1,0,0,22) b.BackgroundColor3 = Color3.fromRGB(45,45,45)
		b.TextColor3 = Color3.fromRGB(220,220,220) b.Text = "  "..cn b.TextXAlignment = Enum.TextXAlignment.Left
		b.Font = Enum.Font.SourceSans b.TextSize = 13
		b.MouseButton1Click:Connect(function()
			clearMenus()
			local ok, obj
			if cn == "Rainbow" then
				ok = true
				obj = Instance.new("Frame")
				obj.Name = "Rainbow"
				obj.Size = UDim2.new(0,200,0,100) obj.Position = UDim2.new(0,20,0,20)
				obj.BorderSizePixel = 0 obj.BackgroundColor3 = Color3.fromRGB(255,255,255)
				connectRB(obj) makeDrag(obj)
			else ok, obj = pcall(function() return Instance.new(cn) end) end
			if ok and obj then
				obj.Name = cn
				if obj:IsA("ScreenGui") then
					obj.IgnoreGuiInset = true obj.ResetOnSpawn = false obj.Parent = UserFolder
				else
					if obj:IsA("GuiObject") and not obj:IsA("UIGradient") and not obj:IsA("UIStroke") and not obj:IsA("UICorner") and not obj:IsA("UIPadding") and not obj:IsA("UIAspectRatioConstraint") and not obj:IsA("UIListLayout") and not obj:IsA("UIGridLayout") then
						obj.Size = UDim2.new(0,120,0,60) obj.Position = UDim2.new(0,20,0,20) makeDrag(obj)
					end
					obj.Parent = parent
				end
				renderTree()
			end
		end)
	end
	activeMenu = mf
end

local expStates = {[UserFolder] = true}
local function createNode(t, indent, isRoot)
	indent = indent or 0
	local it = Instance.new("Frame", scroll)
	it.Size = UDim2.new(1,0,0,22) it.BackgroundTransparency = 1
	local nb = Instance.new("TextButton", it)
	nb.Size = UDim2.new(1,-(indent*15+24),1,0) nb.Position = UDim2.new(0, indent*15+24, 0, 0)
	nb.BackgroundColor3 = isRoot and Color3.fromRGB(70,70,70) or Color3.fromRGB(55,55,55)
	nb.BackgroundTransparency = 0.5 nb.TextColor3 = Color3.fromRGB(240,240,240) nb.Text = ""
	nb.TextXAlignment = Enum.TextXAlignment.Left nb.Font = Enum.Font.SourceSans nb.TextSize = 12
	nb.ClipsDescendants = true
	if selObj == t then
		selNode = nb
		nb.BackgroundColor3 = Color3.fromRGB(0,120,215) nb.BackgroundTransparency = 0.2
	end
	local lbl = Instance.new("TextLabel", nb)
	lbl.Size = UDim2.new(1,-4,1,0) lbl.Position = UDim2.new(0,4,0,0)
	lbl.BackgroundTransparency = 1
	lbl.TextColor3 = Color3.fromRGB(240,240,240)
	lbl.TextXAlignment = Enum.TextXAlignment.Left
	lbl.Font = Enum.Font.SourceSans lbl.TextSize = 12
	lbl.ClipsDescendants = true lbl.TextTruncate = Enum.TextTruncate.AtEnd
	if isRoot then
		lbl.Text = "My Workspace"
	else
		lbl.Text = t.Name.." ("..t.ClassName..")"
	end
	nb.MouseButton1Click:Connect(function() selectObj(t, nb) end)
	local ab = Instance.new("TextButton", it)
	ab.Size = UDim2.new(0,20,1,0) ab.Position = UDim2.new(0, indent*15, 0, 0)
	ab.BackgroundColor3 = Color3.fromRGB(0,120,215) ab.TextColor3 = Color3.fromRGB(255,255,255)
	ab.Font = Enum.Font.SourceSansBold ab.TextSize = 13 ab.Text = "+"
	ab.MouseButton1Click:Connect(function()
		local ap, as = ab.AbsolutePosition, ab.AbsoluteSize
		showMenu(t, UDim2.new(0, ap.X + as.X + 5, 0, ap.Y))
	end)
	local ch = {}
	for _, c in ipairs(t:GetChildren()) do
		if c.Name ~= "SelectionBoxOverlay" and c.Name ~= "DragScript" and c.Name ~= "RainbowScript" then
			table.insert(ch, c)
		end
	end
	if expStates[t] == nil then expStates[t] = true end
	if #ch > 0 and expStates[t] then
		for _, c in ipairs(ch) do createNode(c, indent+1, false) end
	end
end

renderTree = function()
	for _, c in ipairs(scroll:GetChildren()) do if c:IsA("Frame") then c:Destroy() end end
	createNode(UserFolder, 0, true)
end

for _, d in ipairs(UserFolder:GetDescendants()) do
	if d:GetAttribute("RainbowEnabled") and d:IsA("GuiObject") then connectRB(d) end
end

saveState()
renderTree()
