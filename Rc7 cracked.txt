local function isSuspicious(name)
    local lowerName = string.lower(tostring(name or ""))
    for _, keyword in ipairs(suspiciousKeywords) do
        if string.find(lowerName, keyword, 1, true) then
            return true
        end
    end
    return false
end

local function scanRemotes()
    suspiciousRemotes = {}
    local containers = {ReplicatedStorage, Workspace, LocalPlayer, CoreGui, Lighting}
    for _, container in pairs(containers) do
        for _, v in pairs(container:GetDescendants()) do
            if v:IsA("RemoteEvent") then
                if isSuspicious(v.Name) then
                    table.insert(suspiciousRemotes, v)
                end
            end
        end
    end
end
